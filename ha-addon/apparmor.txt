# This profile allows the add-on to function correctly while providing a baseline of security.
# It is designed to be compliant with Home Assistant's security model.
#include <tunables/global>

profile gesture-vision flags=(attach_disconnected,complain) {
  #include <abstractions/base>
  #include <abstractions/node>
  #include <abstractions/nameservice>
  #include <abstractions/user-tmp>

  # Allow access to video devices
  /dev/video* rw,

  # Allow all standard network access for Node.js, Nginx, and MediaMTX
  network,

  # Allow execution of the main processes
  /usr/local/bin/mediamtx ixr,
  /usr/local/bin/docker-entrypoint.sh ixr,
  /usr/sbin/nginx ixr,

  # Allow read/write access to persistent data and log directories
  /data/** rwk,
  /config/** rwk,
  /tmp/** rwk,
  /var/log/nginx/** rwk,
  /var/lib/nginx/** rwk,
  
  # Allow access to shared libraries needed by Node.js and other binaries
  /lib{,32,64}/** mr,

  # Allow shell execution for runOnReady/NotReady scripts
  /bin/sh ixr,
  /bin/dash ixr,
  /usr/bin/curl ixr,
  
  # Capabilities needed by MediaMTX and Nginx
  capability net_bind_service,
  capability chown,
  capability setgid,
  capability setuid,
  capability dac_override,

  # Allow signal handling for process management
  signal,
}
